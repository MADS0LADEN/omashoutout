cmake_minimum_required(VERSION 3.20)
project(shoutout_kde LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
find_package(KF6KCMUtils REQUIRED)
find_package(KF6CoreAddons REQUIRED)
add_library(kcm_shoutout MODULE kcm.cpp)
set_target_properties(kcm_shoutout PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../bin")
target_link_libraries(kcm_shoutout PRIVATE Qt6::Core Qt6::Widgets Qt6::Network KF6::KCMUtils KF6::CoreAddons)
install(TARGETS kcm_shoutout DESTINATION lib/qt6/plugins/plasma/kcms/systemsettings_qwidgets)
add_executable(kde_smoke smoke.cpp)
target_link_libraries(kde_smoke PRIVATE Qt6::Core Qt6::Widgets Qt6::Network KF6::KCMUtils KF6::CoreAddons)
