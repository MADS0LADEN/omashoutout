cmake_minimum_required(VERSION 3.20)
project(omashoutout_kde LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets Network)
find_package(KF6KCMUtils REQUIRED)
find_package(KF6CoreAddons REQUIRED)
add_library(kcm_omashoutout MODULE kcm.cpp)
set_target_properties(kcm_omashoutout PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/..")
target_link_libraries(kcm_omashoutout PRIVATE Qt6::Core Qt6::Widgets Qt6::Network KF6::KCMUtils KF6::CoreAddons)
install(TARGETS kcm_omashoutout DESTINATION lib/qt6/plugins/plasma/kcms/systemsettings_qwidgets)
add_executable(kde_smoke smoke.cpp)
target_link_libraries(kde_smoke PRIVATE Qt6::Core Qt6::Widgets Qt6::Network KF6::KCMUtils KF6::CoreAddons)
